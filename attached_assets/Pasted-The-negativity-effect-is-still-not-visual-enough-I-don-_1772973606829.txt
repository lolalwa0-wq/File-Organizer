// === WAX NEGATIVITY PARTICLE SYSTEM ===

function initWaxNegativity() {
  const waxButton = document.querySelector('.wax-cta');
  if (!waxButton) return;

  // Create dark particles container
  const particleContainer = document.createElement('div');
  particleContainer.classList.add('negativity-particles');
  particleContainer.style.cssText = `
    position: absolute;
    inset: 0;
    pointer-events: none;
    z-index: 4;
    overflow: visible;
    border-radius: inherit;
  `;
  waxButton.appendChild(particleContainer);

  // Dark overlay that will reduce as particles leave
  const darkOverlay = document.createElement('div');
  darkOverlay.classList.add('wax-dark-overlay');
  darkOverlay.style.cssText = `
    position: absolute;
    inset: 0;
    z-index: 2;
    pointer-events: none;
    border-radius: inherit;
    background: linear-gradient(
      135deg,
      rgba(15, 0, 25, 0.92) 0%,
      rgba(30, 5, 40, 0.88) 30%,
      rgba(20, 0, 30, 0.9) 60%,
      rgba(10, 0, 20, 0.95) 100%
    );
    transition: opacity 3s ease-in-out;
  `;
  waxButton.appendChild(darkOverlay);

  let isHovering = false;
  let particleInterval = null;
  let cleanupProgress = 0;

  function createDarkParticle() {
    const particle = document.createElement('div');
    const size = Math.random() * 12 + 6; // 6-18px
    const startX = Math.random() * waxButton.offsetWidth;
    const startY = Math.random() * waxButton.offsetHeight;
    const driftX = (Math.random() - 0.5) * 80;
    const duration = Math.random() * 2000 + 1500; // 1.5-3.5s

    // Each particle is a dark blob
    particle.style.cssText = `
      position: absolute;
      left: ${startX}px;
      top: ${startY}px;
      width: ${size}px;
      height: ${size}px;
      border-radius: 50%;
      background: radial-gradient(circle,
        rgba(20, 0, 35, 0.9) 0%,
        rgba(35, 5, 50, 0.7) 40%,
        rgba(15, 0, 25, 0.4) 70%,
        transparent 100%
      );
      pointer-events: none;
      z-index: 5;
      box-shadow: 
        0 0 ${size/2}px rgba(20, 0, 35, 0.6),
        0 0 ${size}px rgba(30, 5, 45, 0.3);
      animation: waxParticleFall ${duration}ms ease-in forwards;
      --drift-x: ${driftX}px;
    `;

    particleContainer.appendChild(particle);

    // Remove particle after animation
    setTimeout(() => {
      particle.remove();
    }, duration);
  }

  // On hover — start releasing dark particles
  waxButton.addEventListener('mouseenter', () => {
    isHovering = true;
    cleanupProgress = 0;

    // Spawn particles gradually — dark blobs floating away
    let spawnCount = 0;
    particleInterval = setInterval(() => {
      if (!isHovering) return;

      // Spawn 2-4 particles at a time
      const batch = Math.floor(Math.random() * 3) + 2;
      for (let i = 0; i < batch; i++) {
        createDarkParticle();
      }

      spawnCount++;
      cleanupProgress = Math.min(spawnCount / 20, 1);

      // Gradually fade the dark overlay as particles leave
      darkOverlay.style.opacity = 1 - cleanupProgress;

      // After enough particles, button is clean
      if (spawnCount > 25) {
        darkOverlay.style.opacity = 0;
        clearInterval(particleInterval);
      }
    }, 120); // new particles every 120ms

    // Button slowly starts glowing gold
    waxButton.style.transition = 'all 3s ease-in-out';
    waxButton.style.borderColor = 'rgba(201, 162, 39, 0.5)';
    waxButton.style.boxShadow = `
      0 0 20px rgba(201, 162, 39, 0.15),
      0 0 50px rgba(201, 162, 39, 0.08)
    `;
  });

  // On mouse leave — negativity creeps back
  waxButton.addEventListener('mouseleave', () => {
    isHovering = false;
    clearInterval(particleInterval);

    // Dark overlay creeps back slowly
    darkOverlay.style.transition = 'opacity 5s ease-in';
    darkOverlay.style.opacity = 1;

    // Button loses its glow
    waxButton.style.transition = 'all 5s ease-in';
    waxButton.style.borderColor = 'rgba(201, 162, 39, 0.15)';
    waxButton.style.boxShadow = 'none';
  });
}

// Initialize when DOM is ready
document.addEventListener('DOMContentLoaded', initWaxNegativity);
